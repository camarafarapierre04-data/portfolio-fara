-- MESURES DAX – créer une table de mesures dans Power BI
Total inspections = COUNTROWS(Inspections)

Inspections conformes =
CALCULATE([Total inspections], Inspections[Statut_Conformite] = "Conforme")

Inspections non conformes =
CALCULATE([Total inspections], Inspections[Statut_Conformite] = "Non conforme")

Taux de conformité =
DIVIDE([Inspections conformes], [Total inspections], 0)

Actions en retard =
CALCULATE([Total inspections], Inspections[Action_Corrective] = "En retard")

Coût total des actions (GNF) =
SUM(Inspections[Cout_Action_GNF])

Coût moyen par action (GNF) =
DIVIDE([Coût total des actions (GNF)],
       CALCULATE([Total inspections], Inspections[Cout_Action_GNF] > 0), 0)
